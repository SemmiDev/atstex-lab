%% \cvrow{left}{right} — a plain two-column row, NO \item, safe outside itemize
\newcommand{\cvrow}[2]{%
  \noindent
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \small #1 & \small #2 \\
  \end{tabular*}\vspace{-2pt}%
}
