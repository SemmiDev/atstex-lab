%% ── Reusable Macros ────────────────────────────────────────────────

%% Two-column job row: bold left / right-aligned date
\newcommand{\cpJobRow}[2]{%
  \noindent
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    #1 & \small #2
  \end{tabular*}%
}

%% Four-field subheading (company, location, title, dates)
\newcommand{\cpSubheading}[4]{%
  \vspace{-1pt}
  \item
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{#1} & \small #2 \\
    \textit{\small #3} & \textit{\small #4} \\
  \end{tabular*}\vspace{-6pt}%
}

%% Project / award heading: left content, right date
%% NOTE: uses \linewidth (not \textwidth) — correct inside itemize with leftmargin
\newcommand{\cpProjectRow}[2]{%
  \item
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \small #1 & \small #2 \\
  \end{tabular*}\vspace{-6pt}%
}

%% Bullet item
\newcommand{\cpItem}[1]{%
  \item\small{#1\vspace{-2pt}}%
}

%% List wrappers
\newcommand{\cpListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\cpListEnd}{\end{itemize}}
\newcommand{\cpBulletStart}{\begin{itemize}[leftmargin=0.2in, label={\tiny$\bullet$}, itemsep=0pt, topsep=2pt, parsep=0pt]}
\newcommand{\cpBulletEnd}{\end{itemize}\vspace{-4pt}}
