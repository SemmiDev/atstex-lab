%% ATSTEXLAB — sky
%% A clean, modern ATS-optimised template with a sky-blue accent system.
%%
%% Design principles:
%%   • 100% ATS-parseable: strict single-column, no floating boxes,
%%     semantic reading order guaranteed.
%%   • Photo-aware header: minipage layout when .Personal.Photo is set.
%%   • Sky-blue accent (skyblue) on name, section titles & rules only —
%%     body text stays black for maximum OCR / parser fidelity.
%%   • Subtle left-border accent on section headings via titlesec
%%     after-code — visually distinct without sacrificing parse order.
%%   • Compact entry macros borrowed from the best of bay + wave.
%%   • Every optional block fully guarded — nothing renders when absent.
%%   • Field names match cvtemplate struct exactly:
%%       Experience/Education/Volunteer  → .Dates  (not StartDate/EndDate)
%%       Awards/Talks                    → .Date
%%       Certifications                  → .Name, .Issuer  (no date)
%%       Skills                          → .Languages .Frameworks .Tools .Other
%%       Projects                        → .Name .Role .Link .Bullets
%%       Volunteer                       → .Organization .Role .Dates .Location .Bullets
%% ══════════════════════════════════════════════════════════════════════

%% ── Reusable Macros ────────────────────────────────────────────────

%% \skySubheading{org}{location}{role/degree}{dates}
%% Four-field entry used for Experience, Education, Volunteer
\newcommand{\skySubheading}[4]{%
  \vspace{2pt}\item
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{\color{darkgrey}#1} & \small\textit{\color{midgrey}#2} \\
    \textit{\small\color{darkgrey}#3} & \textit{\small\color{midgrey}#4} \\
  \end{tabular*}\vspace{-2pt}%
}

%% \skyProjectRow{left content}{right date}
\newcommand{\skyProjectRow}[2]{%
  \vspace{2pt}\item
  \begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \small #1 & \small\textit{\color{midgrey}#2} \\
  \end{tabular*}\vspace{-2pt}%
}

%% Bullet item
\newcommand{\skyItem}[1]{%
  \item {\small\color{darkgrey}#1}\vspace{0pt}%
}

%% List wrappers
\newcommand{\skyListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\skyListEnd}{\end{itemize}}
\newcommand{\skyBulletStart}{%
  \begin{itemize}[%
    leftmargin=0.20in,%
    label={\textcolor{skyblue}{\tiny$\blacksquare$}},%
    itemsep=1pt,%
    topsep=3pt,%
    parsep=1pt%
  ]%
}
\newcommand{\skyBulletEnd}{\end{itemize}\vspace{-2pt}}
